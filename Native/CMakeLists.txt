cmake_minimum_required(VERSION 3.20)
project(XbControllerNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 専用です。")
endif()
set(GAMEINPUT_SDK "" CACHE PATH "Microsoft.GameInput NuGet の展開先")
if(NOT EXISTS "${GAMEINPUT_SDK}/native/include/GameInput.h")
  message(FATAL_ERROR "GAMEINPUT_SDK を設定してください。")
endif()
add_library(XbControllerNative SHARED bridge.cpp)
target_compile_features(XbControllerNative PRIVATE cxx_std_17)
target_include_directories(XbControllerNative PRIVATE "${GAMEINPUT_SDK}/native/include")
target_link_libraries(XbControllerNative PRIVATE "${GAMEINPUT_SDK}/native/lib/x64/GameInput.lib" advapi32 version)
target_compile_definitions(XbControllerNative PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(XbControllerNative PRIVATE /W4 /WX /permissive- /guard:cf /utf-8)
target_link_options(XbControllerNative PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
set_property(TARGET XbControllerNative PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_executable(XbControllerRawProbe raw_probe.cpp)
target_compile_features(XbControllerRawProbe PRIVATE cxx_std_17)
target_include_directories(XbControllerRawProbe PRIVATE "${GAMEINPUT_SDK}/native/include")
target_link_libraries(XbControllerRawProbe PRIVATE "${GAMEINPUT_SDK}/native/lib/x64/GameInput.lib" advapi32 version)
target_compile_definitions(XbControllerRawProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(XbControllerRawProbe PRIVATE /W4 /WX /permissive- /utf-8)
set_property(TARGET XbControllerRawProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(BUILD_TESTING)
  add_executable(XbControllerNativeTests tests.cpp)
  target_compile_features(XbControllerNativeTests PRIVATE cxx_std_17)
  target_include_directories(XbControllerNativeTests PRIVATE "${GAMEINPUT_SDK}/native/include")
  target_link_libraries(XbControllerNativeTests PRIVATE "${GAMEINPUT_SDK}/native/lib/x64/GameInput.lib" advapi32 version)
  target_compile_definitions(XbControllerNativeTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(XbControllerNativeTests PRIVATE /W4 /WX /permissive- /utf-8)
  set_property(TARGET XbControllerNativeTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME native-regressions COMMAND XbControllerNativeTests)
endif()
